============================================================
API PIC2D (PICO-8–подобный для Lua)
============================================================

ГРАФИКА И ТЕКСТ:
------------------------------------------------------------
cls(color)                  – очистить экран цветом 0..15
spr(n, x, y)                – спрайт n (8x8) в (x,y)
spr(n, x, y, w, h, fx, fy)  – w,h кратность (1=8x8,2=16x16...),
                              fx/fy – true/false для отражения
spr_scale(n, x, y, zoom)    – спрайт n с масштабом zoom (целое)
print(text, x, y, color)    – вывести текст в (x,y) цветом color

ПРИМИТИВЫ:
------------------------------------------------------------
circ(x, y, r, color)        – контур круга
circfill(x, y, r, color)    – залитый круг
rect(x1,y1,x2,y2,color)     – контур прямоугольника
rectfill(x1,y1,x2,y2,color) – залитый прямоугольник

ВВОД (КЛАВИАТУРА):
------------------------------------------------------------
btn(key)   – true если кнопка зажата
btnp(key)  – true если кнопка нажата в этом кадре
Коды: 0=влево, 1=вправо, 2=вверх, 3=вниз, 4=Z/J, 5=X/K

КАМЕРА:
------------------------------------------------------------
camera(x, y) – установить смещение камеры (x,y),
               возвращает старые координаты: old_x, old_y = camera(x,y)

МЫШЬ:
------------------------------------------------------------
mouse()   – возвращает таблицу { x, y, left, middle, right, scrollx, scrolly }
mousep()  – аналогично, но кнопки только при нажатии (как btnp)

ЗВУК:
------------------------------------------------------------
sfx(index) – проиграть звук с индексом index (0..63)

СОХРАНЕНИЕ:
------------------------------------------------------------
save(pos, value) – сохранить 64-битное число в слот pos (0..63),
                   данные записываются в файл только при закрытии
                   приложения или автосохранении
load(pos)        – загрузить число из слота pos (0, если нет)
is_save()        – true, если файл сохранения storage.bin существует

КАРТА (ТАЙЛЫ):
------------------------------------------------------------
map_draw(celx, cely, sx, sy, celw, celh) – рисует участок карты
mget(x, y)         – возвращает ID тайла в (x,y)
mset(x, y, id)     – устанавливает ID тайла в (x,y)

Примечание: Флагов нет — вместо них используйте таблицы Lua для хранения свойств объектов.
-- Свойства тайлов
local tile_props = {
    [59] = { gravel = true, speed_mult = 1.0 },
    [60] = { gravel = false, speed_mult = 2.0 },
    [1]  = { solid = true },
}

ОСОБЕННОСТИ:
------------------------------------------------------------
- Экран 128x128, 16 цветов (палитра PICO-8), поддержка карта 256x256 в формате .csv
- Игра должна определять функции: _init(), _update(), _draw()
- Запуск: pic2d mygame.pic (ZIP с main.lua и ресурсами)
- Ресурсы в архиве: assets/ (spritesheet, font) и sounds/ (0.wav ...)



Пример:

-- main.lua
function init()
    cls(12)
end

function update()
    -- пусто
end

function draw()
    cls(12)
    print("HELLO FROM PICOLIB-LUA!", 10, 20, 7)
    -- рисуем спрайт номер 1 (если есть в sprites.png)
    spr(1, 60, 60)
end
